cmake_minimum_required(VERSION 3.24)
project(session_occt_reference LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OCCT_INCLUDE_DIR "" CACHE PATH "OCCT headers for an existing static build")
set(OCCT_LIBRARY_DIR "" CACHE PATH "OCCT libraries for an existing static build")
set(GEOMETRY_FIXTURES "${CMAKE_CURRENT_SOURCE_DIR}/../../build/geometry-tests" CACHE PATH "Exported C++ regression geometry")

add_executable(occt_reference main.cpp)
target_include_directories(occt_reference PRIVATE ../../src/json/include)
target_compile_definitions(occt_reference PRIVATE GEOMETRY_FIXTURES="${GEOMETRY_FIXTURES}")

set(OCCT_LIBRARIES TKOffset TKFillet TKBool TKBO TKShHealing TKPrim TKTopAlgo TKGeomAlgo TKBRep TKGeomBase TKG3d TKG2d TKMath TKernel)
if(OCCT_INCLUDE_DIR AND OCCT_LIBRARY_DIR)
    target_include_directories(occt_reference SYSTEM PRIVATE "${OCCT_INCLUDE_DIR}")
    target_compile_definitions(occt_reference PRIVATE OCCT_STATIC_BUILD)
    foreach(library IN LISTS OCCT_LIBRARIES)
        find_library(${library}_PATH NAMES ${library} PATHS "${OCCT_LIBRARY_DIR}" NO_DEFAULT_PATH REQUIRED)
        target_link_libraries(occt_reference PRIVATE "${${library}_PATH}")
    endforeach()
else()
    find_package(OpenCASCADE REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms)
    target_include_directories(occt_reference SYSTEM PRIVATE "${OpenCASCADE_INCLUDE_DIR}")
    target_link_libraries(occt_reference PRIVATE ${OCCT_LIBRARIES})
endif()

if(WIN32)
    target_compile_definitions(occt_reference PRIVATE NOMINMAX)
    target_link_libraries(occt_reference PRIVATE ws2_32 psapi advapi32 user32)
endif()
